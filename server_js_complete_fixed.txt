app.use(express.static(path.join(__dirname, '../'), {
  setHeaders: (res, path) => {
    if (path.includes('.js') || path.includes('.css')) {
      res.setHeader('Cache-Control', 'no-cache');
    }
    if (path.includes('app.js')) {
      res.setHeader('Cache-Control', 'no-cache, no-store, must-revalidate');
      res.setHeader('Pragma', 'no-cache');
      res.setHeader('Expires', '0');
    }
    if (path.includes('manifest.json') || path.includes('sw.js')) {
      res.setHeader('Cache-Control', 'no-cache, no-store, must-revalidate');
    }
  }
}));

app.post('/api/setup-schema', async (req, res) => {
  console.log('🔧 Schema setup requested...');
  
  try {
    const { Pool } = require('pg');
    
    const pool = new Pool({
      user: process.env.DB_USER,
      host: process.env.DB_HOST,
      database: process.env.DB_NAME,
      password: process.env.DB_PASSWORD,
      port: process.env.DB_PORT,
      ssl: process.env.DB_SSL && process.env.DB_SSL !== 'false' ? {
        rejectUnauthorized: false
      } : false
    });

    const client = await pool.connect();
    
    try {
      // Check if customer_id column exists
      const columnCheck = await client.query(`
        SELECT column_name 
        FROM information_schema.columns 
        WHERE table_name = 'customers' AND column_name = 'customer_id'
      `);
      
      if (columnCheck.rows.length === 0) {
        console.log('Adding customer_id column...');
        await client.query('ALTER TABLE customers ADD COLUMN customer_id VARCHAR(100) UNIQUE');
        
        // Update existing customers with unique customer_id
        const existingCustomers = await client.query('SELECT id FROM customers WHERE customer_id IS NULL');
        for (const customer of existingCustomers.rows) {
          const customerId = 'CUST_' + Date.now() + '_' + Math.random().toString(36).substr(2, 5);
          await client.query('UPDATE customers SET customer_id = $1 WHERE id = $2', [customerId, customer.id]);
        }
      }
      
      console.log('✅ Schema setup completed');
      
      res.json({
        success: true,
        message: 'Schema setup completed successfully'
      });
      
    } finally {
      client.release();
      await pool.end();
    }
    
  } catch (error) {
    console.error('❌ Schema setup failed:', error);
    res.status(500).json({
      success: false,
      message: 'Schema setup failed',
      error: error.message
    });
  }
});
